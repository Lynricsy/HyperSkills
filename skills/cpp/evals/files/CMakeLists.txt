cmake_minimum_required(VERSION 3.20)
project(packet_fixture LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(packet STATIC packet.cpp)
target_include_directories(packet PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(packet PUBLIC cxx_std_17)
target_compile_definitions(packet PRIVATE PACKET_EXTENDED=1)
add_executable(packet_client packet-client.cpp)
target_link_libraries(packet_client PRIVATE packet)

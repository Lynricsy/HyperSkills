== .ci/deploy-job.sh (runs on the self-hosted runner) ==
#!/usr/bin/env bash
set -euo pipefail
docker compose up -d --build --wait web app
curl -fsS http://127.0.0.1:8080/healthz.txt
timeout 420 docker compose run --rm terraform
docker compose down

== infra/deploy.sh (entrypoint of the terraform service, image FROM hashicorp/terraform:1.15 = Alpine) ==
#!/bin/sh
set -eu
terraform init -input=false
terraform apply -input=false -auto-approve
echo "apply finished"

== infra/backend.tf ==
terraform {
  backend "s3" {
    bucket       = "acme-tfstate"
    key          = "shop/prod.tfstate"
    region       = "eu-central-1"
    use_lockfile = true
  }
}

== docker/nginx.conf (excerpt) ==
location = /healthz.txt { root /srv/static; }
location ~ \.php$ {
    fastcgi_pass unix:/run/php-fpm/php-fpm.sock;
    include fastcgi_params;
}
